function Write-Log {

    param(

        [string]$Message,

        [ValidateSet("INFO","WARNING","ERROR","SUCCESS")]
        [string]$Level="INFO"

    )

    $Time = Get-Date -Format "yyyy-MM-dd HH:mm:ss"

    switch ($Level) {

        "INFO"     { $Color="White" }
        "WARNING"  { $Color="Yellow" }
        "ERROR"    { $Color="Red" }
        "SUCCESS"  { $Color="Green" }

    }

    Write-Host "[$Time] [$Level] $Message" -ForegroundColor $Color

}
Export-ModuleMember -Function Write-Log
